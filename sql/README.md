users/ - Сначала создаются пользователи/роли
schemas/ - Затем схемы с указанием владельцев
tables/ - Пустые таблицы (только CREATE TABLE)
data/ - Загрузка начальных данных
indexes/ - Создание индексов на заполненных таблицах
constraints/ - Внешние ключи между уже существующими таблицами

--------------------------------------------------------------------------------------------------

db_helper.sh позволяет работать с таблицами (можно загружать отдельные скрипты/директории)

echo "Использование: db_helper.sh [СТРОКА ПОДКЛЮЧЕНИЯ К БД] [ОПЦИЯ] [ПАРАМЕТРЫ ОПЦИИ]"
    echo ""
    echo "Опции:"
    echo "  -h, --help        Показать справку"
    echo "  -l, --list        Показать доступные директории"
    echo "  -f, --file        Выполнить SQL-скрипт"
    echo "  -d, --dir         Выполнить SQL-скрипты в указанной директории"
    echo "  -b, --build       Собрать базу данных"
    echo ""

Пример:

Загрузить скрипт ./schemas/schema.sql
./db_helper.sh postgres://arms_admin:arms_qwerty@localhost:5432/arms -f ./schemas/schema.sql

Загрузить все скрипты из директории ./tables
./db_helper.sh postgres://arms_admin:arms_qwerty@localhost:5432/arms -d ./tables

Полностью пересобрать БД
./db_helper.sh postgres://arms_admin:arms_qwerty@localhost:5432/arms -b

--------------------------------------------------------------------------------------------------

Перед выполнением скриптов необходимо создать ссылку на .env файл в папке, в которой выполняется скрипт.
Команда для этого:
ln -s ../../.env .env

Также возможно необходимо выдать права на скрипт
chmod +x ./db_helper.sh
